import os
import uuid
from flask import Flask, request, jsonify, send_file
import yt_dlp

app = Flask(__name__)

DOWNLOAD_DIR = "/tmp/tiktok_downloads"
os.makedirs(DOWNLOAD_DIR, exist_ok=True)


@app.route("/")
def home():
    return "TikTok Downloader API is running!"


@app.route("/download", methods=["GET"])
def download():
    url = request.args.get("url", "").strip()

    if not url:
        return jsonify({"error": "Missing url"}), 400

    filename = str(uuid.uuid4())
    output = os.path.join(DOWNLOAD_DIR, filename + ".%(ext)s")

    options = {
        "outtmpl": output,
        "format": "best[ext=mp4]/best",
        "noplaylist": True,
        "quiet": True,
        "no_warnings": True,
        "retries": 3,
    }

    try:
        with yt_dlp.YoutubeDL(options) as ydl:
            info = ydl.extract_info(url, download=True)
            downloaded = ydl.prepare_filename(info)

        if not os.path.exists(downloaded):
            base = os.path.splitext(downloaded)[0]

            for name in os.listdir(DOWNLOAD_DIR):
                if name.startswith(os.path.basename(base)):
                    downloaded = os.path.join(DOWNLOAD_DIR, name)
                    break

        if not os.path.exists(downloaded):
            return jsonify({"error": "Download failed"}), 500

        return send_file(
            downloaded,
            mimetype="video/mp4",
            as_attachment=True,
            download_name="tiktok.mp4",
        )

    except Exception as e:
        print("DOWNLOAD ERROR:", repr(e))
        return jsonify({
            "error": "Could not download video",
            "details": str(e)
        }), 500


if __name__ == "__main__":
    port = int(os.environ.get("PORT", 10000))
    app.run(host="0.0.0.0", port=port)
